//https://leetcode.com/problems/add-two-numbers
// Time and space complexity is O(max(n, m))
class Solution {
    public ListNode addTwoNumbers(ListNode l1, ListNode l2) {
        ListNode cur1 = l1;
        ListNode cur2 = l2;
        ListNode newListHead = null;
        ListNode current = null;
        int result = 0;
        int sum = 0;
        int carry = 0;
        while(cur1!=null || cur2!=null ||carry!=0){
            int val1 = 0;
            int val2 = 0;
            if(cur1 != null){
                val1 = cur1.val;
            }

            if(cur2 != null){
                val2 = cur2.val;
            }

            // sum = cur1.val+cur2.val+carry;
            sum = val1+val2+carry;
            if(sum < 10){
                result = sum;
                carry = 0;
            }else{
                result = sum%10;
                carry = sum/10;
            }
            ListNode newNode = new ListNode(result);
            if(newListHead == null){
                newListHead = newNode;
                current = newNode;
            }else{
                current.next = newNode;
                current = newNode;
            }
            
            // cur1 = cur1.next;
            // cur2 = cur2.next;
            if(cur1 != null)
            {
                cur1 = cur1.next;
            }

            if(cur2 != null)
            {
                cur2 = cur2.next;
            }
        }        
        return newListHead;    
    }
}
