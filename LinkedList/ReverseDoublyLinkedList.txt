timecomplexity--->  O(n)
class Solution {
    public Node reverse(Node head) {
        // code here
        Node cur = head;
        Node tempNode;
       
        while(cur!=null){
            tempNode = cur.next;
            cur.next = cur.prev;
            cur.prev = tempNode;
            cur = tempNode;
        }
        
        
        // second loop too update head to new head
        while(head.prev!=null){
            head = head.prev;
        }
        return head;
    }
}
