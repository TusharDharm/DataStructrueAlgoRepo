    // This is optimized with TC->O(n) and SC-> O(1)
     public ListNode mergeNodes(ListNode head) {
      ListNode current = head;
      ListNode cur = head.next;
      int sum = 0;
      while(cur!=null){
        if(cur.val == 0){
            current = current.next;
            current.val = sum;      
            sum = 0;
        }else{
            sum += cur.val;
        }
        cur = cur.next;
      }
      current.next = null;
      return head.next;  
    }
}
