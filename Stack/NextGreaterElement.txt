//TC --> O(n) Sc--> O(n)
class Solution {
    public ArrayList<Integer> nextLargerElement(int[] arr) {

        Stack<Integer> s = new Stack<>();
        int n = arr.length;

        ArrayList<Integer> ans = new ArrayList<>();

        for (int i = n - 1; i >= 0; --i) {

            while (!s.empty() && s.peek() <= arr[i]) {
                s.pop();
            }

            ans.add(s.empty() ? -1 : s.peek());

            s.push(arr[i]);
        }

        Collections.reverse(ans);

        return ans;
    }
}
